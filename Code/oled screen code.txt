#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET -1
#define OLED_ADDRESS 0x3C
#define OLED_SDA 17
#define OLED_SCL 18

Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

void setup() {
  Wire.begin(OLED_SDA, OLED_SCL);

  if (!display.begin(SSD1306_SWITCHCAPVCC, OLED_ADDRESS)) {
    // Stop here if the display cannot be initialized.
    while (true) {
      delay(100);
    }
  }

  display.clearDisplay();
  // Text size is a whole-number scale (1 = 6x8 px per character); 1.5 was truncated to 1.
  // Size 2 would make "Hello World!" wider than the 128 px screen.
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0, 24);
  display.println(F("Hello World!"));
  display.display();
}

void loop() {
  // The greeting remains on screen.
}
